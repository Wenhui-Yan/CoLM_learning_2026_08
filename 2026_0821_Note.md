# ============================================================
# 注册一个新数据
# ============================================================
1.扫描参考数据目录
# 扫描 Reference 目录，看看有哪些数据可以注册
openbench ref scan C:/Users/Administrator/Desktop/openbench/Reference/Grid/MidRes/ --dry-run
# 或者扫描整个 Reference 目录
openbench ref scan C:/Users/Administrator/Desktop/openbench/Reference/ --dry-run
2.注册 GLEAM 蒸散发数据集
openbench ref register GLEAM_v4.3_LowRes \
    --root-dir C:/Users/Administrator/Desktop/openbench/Reference/Grid/MidRes/ \
    --data-type grid \
    --grid-res 0.25 \
    --tim-res Day \
    --category Water \
    --years 1980 2022 \
    -v 'Evapotranspiration name=E unit="mm day-1" sub_dir=Water/Evapotranspiration/GLEAM_v4.3 prefix=E_ suffix=_GLEAM_v4.3'
3.用注册后的数据进行评估
